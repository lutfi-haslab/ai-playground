- **Subtotal**: Sum of `price * quantity` for all items, rounded to 2 decimals.
- **Discount**:
  - `SAVE10` → 10% off on subtotal.

- **Tax**: Applied on the discounted subtotal using the provided `taxRate` (default 8%).
- **Total**: `subtotal - discount + tax`.
- **Rounding**: All values are rounded to 2 decimal places using a safe `round2` helper to avoid floating‑point precision issues.

/**
 * Calculates cart totals applying optional coupon discounts and a sales tax rate.
 *
 * @param items - List of cart items (price and quantity).
 * @param couponCode - Optional coupon code ('SAVE10' for 10% off, 'FLAT5' for $5 off).
 * @param taxRate   - Sales tax rate as a decimal (e.g. 0.08 for 8%).
 * @returns A CartCalculation object with subtotal, discount, tax, and total (rounded to 2 decimals).
 */
export function calculateCartTotal(
  items: CartItem[],
  couponCode: string = '',
  taxRate: number = 0
): CartCalculation {
  // Compute subtotal
  const subtotal = items.reduce(
    (sum, item) => sum + item.price * item.quantity,
    0
  );

  // Determine discount based on coupon code
  let discount = 0;
  if (couponCode === 'SAVE10') {
    discount = subtotal * 0.1; // 10% off
  } else if (couponCode === 'FLAT5') {
    discount = 5; // $5 off
  }

  // Discount cannot exceed subtotal
  discount = Math.min(discount, subtotal);
  discount = Math.round(discount * 100) / 100;

  // Tax applied to the discounted subtotal
  const taxableAmount = subtotal - discount;
  const tax = Math.round(taxableAmount * taxRate * 100) / 100;

  // Total amount due
  const total = Math.round((taxableAmount + tax) * 100) / 100;

  return {
    subtotal: Math.round(subtotal * 100) / 100,
    discount,
    tax,
    total
  };
}
